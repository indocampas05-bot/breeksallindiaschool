/* ---------- Contact form ---------- */
const form = document.getElementById("contactForm"), phone = document.getElementById("phone"), email = document.getElementById("email");
function validate(){
  let ok = true;
  form.querySelectorAll("[required]").forEach(f => {
    let valid = f.value.trim() !== "";
    if (f === phone) valid = /^(\+91[\s-]?)?[6-9]\d{9}$/.test(f.value.replace(/\s/g,""));
    if (f === email) valid = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(f.value.trim());
    if (f.minLength > 0 && f.value.trim().length < f.minLength) valid = false;
    f.classList.toggle("is-invalid", !valid); f.classList.toggle("is-valid", valid);
    if (!valid) ok = false;
  });
  return ok;
}
form.addEventListener("input", e => { if (e.target.classList.contains("is-invalid")) validate(); });
form.addEventListener("submit", async e => {

  const box = document.getElementById("formAlert"); box.className = "alert mt-4 d-none";

  const btn = document.getElementById("submitBtn"), sp = btn.querySelector(".spinner-border"), lbl = btn.querySelector(".btn-label");
  btn.disabled = true; sp.classList.remove("d-none"); lbl.textContent = "Submitting...";
  const data = {student:student.value, parent:parent.value, phone:phone.value, email:email.value, subject:cls.value, message:msg.value};

/* ---------- Faculty: horizontal scroll (arrows, drag, progress) ---------- */
(function(){
  const t = document.getElementById("tTrack"); if(!t) return;
  const prev = document.getElementById("tPrev"), next = document.getElementById("tNext"), prog = document.getElementById("tProg");
  const step = () => { const c = t.querySelector(".tcard"); return c ? (c.offsetWidth + 18) * 2 : 300; };
  function upd(){
    const max = t.scrollWidth - t.clientWidth;
    prev.disabled = t.scrollLeft <= 2; next.disabled = t.scrollLeft >= max - 2;
    const w = Math.max(12, t.clientWidth / t.scrollWidth * 100);
    prog.style.width = w + "%";
    prog.style.marginLeft = (max > 0 ? t.scrollLeft / max * (100 - w) : 0) + "%";
  }
  prev.addEventListener("click", () => t.scrollBy({left:-step(), behavior:"smooth"}));
  next.addEventListener("click", () => t.scrollBy({left: step(), behavior:"smooth"}));
  t.addEventListener("scroll", upd, {passive:true}); addEventListener("resize", upd); upd();
  /* mouse drag (touch uses native swipe) */
  let down=false, sx=0, sl=0, moved=false;
  t.addEventListener("pointerdown", e => { if(e.pointerType!=="mouse") return; down=true; moved=false; sx=e.clientX; sl=t.scrollLeft; });
  addEventListener("pointermove", e => { if(!down) return; const dx=e.clientX-sx; if(Math.abs(dx)>4){moved=true; t.classList.add("dragging");} t.scrollLeft = sl - dx; });
  addEventListener("pointerup", () => { if(!down) return; down=false; t.classList.remove("dragging"); });
})();
